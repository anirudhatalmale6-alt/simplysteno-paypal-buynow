<!--
  SimplySteno — PayPal Buy Now buttons + mandatory checkbox gate
  ---------------------------------------------------------------
  Paste this whole block into ONE Elementor "HTML" widget, placed at the
  BOTTOM of the payments page (below the last price card).

  Nothing else on the page needs to change. This script finds the existing
  price cards, strips the dead PayPal links, and drops a real PayPal button
  into each one.

  BEFORE GOING LIVE: replace PAYPAL_CLIENT_ID below with the Live client ID.
-->

<script>
(function () {
  "use strict";

  /* ------------------------------------------------------------------ *
   * 1. CONFIG — the only line you normally need to touch.
   *
   *    SANDBOX (fake money, for testing)  <-- currently active
   *    LIVE    (real money)               <-- swap in when you're happy
   *
   *    To go live: comment out the SANDBOX line, uncomment the LIVE line.
   * ------------------------------------------------------------------ */
  var PAYPAL_CLIENT_ID =
    /* SANDBOX */ "AfnC9p0Vcdgx8SSWAx-Z9Tj_IF_ClNjcmb0E5yjqh_Auu6EmsoD_rDRlF-Ja9i5S_84kzJLXv4mmmwwy";
    /* LIVE    */ // "AY1XMO_TbgDuGrqwOewBRgCriJngfl1h0GQrdpVY5n1opz6pLYu-R2EfwerSVXTbsA2sV-uYuM0DlEvL";

  var CURRENCY         = "USD";
  var LOCALE           = "en_US";   // pin it, or PayPal guesses from the
                                    // shopper's IP and shows German/SEPA etc.

  /* ------------------------------------------------------------------ *
   * THE THANK-YOU MESSAGE the customer sees after paying.
   * Edit the text between the quotes. Keep the quotes themselves.
   * Use \n where you want a line break.
   * ------------------------------------------------------------------ */
  var THANK_YOU_MESSAGE =
    "Expect an email within 24 hours with a list of questions. " +
    "You'll need to reply before your account can be set up.";

  /* Which payment buttons appear on each card. Each card shows:
       - the gold "PayPal / Buy Now" button
       - "Pay Later" (PayPal Credit), which the page already advertises
       - a black "Debit or Credit Card" button, for shoppers with no PayPal account
     Everything else (SEPA, iDEAL, giropay, Venmo...) is switched off — they're
     either European-only or irrelevant here, and they just clutter the card. */
  var DISABLE_FUNDING  = "sepa,bancontact,eps,giropay,ideal,mybank,p24,sofort,venmo";

  /* ------------------------------------------------------------------ *
   * 2. Read the price straight off each card's own heading.
   *    "$1,590 (12 months)" -> 1590.00
   *    This means Marc can edit a price in Elementor and the button
   *    follows automatically — no amounts hardcoded here to drift.
   * ------------------------------------------------------------------ */
  function priceFrom(text) {
    var m = String(text).replace(/ /g, " ").match(/\$\s*([\d,]+(?:\.\d{1,2})?)/);
    if (!m) return null;
    var n = parseFloat(m[1].replace(/,/g, ""));
    return (isFinite(n) && n > 0) ? n.toFixed(2) : null;
  }

  function cleanName(text) {
    return String(text).replace(/\s+/g, " ").trim().slice(0, 127);
  }

  /* ------------------------------------------------------------------ *
   * 3. The three mandatory checkboxes.
   * ------------------------------------------------------------------ */
  function boxes() {
    return Array.prototype.slice.call(
      document.querySelectorAll('.checkbox_wrapper input[type="checkbox"]')
    );
  }
  function allTicked() {
    var b = boxes();
    return b.length > 0 && b.every(function (c) { return c.checked; });
  }

  /* Warning shown when someone clicks Pay with boxes unticked. */
  var warning;
  function warnEl() {
    if (warning) return warning;
    warning = document.createElement("p");
    warning.className = "ss-paypal-warning";
    warning.setAttribute("role", "alert");
    warning.textContent =
      "Please tick all three boxes above before continuing to PayPal.";
    warning.style.display = "none";
    var wrap = document.querySelector(".checkbox_wrapper");
    if (wrap) wrap.appendChild(warning);
    return warning;
  }
  function showWarning() {
    var w = warnEl();
    w.style.display = "block";
    w.scrollIntoView({ behavior: "smooth", block: "center" });
  }
  function hideWarning() {
    if (warning) warning.style.display = "none";
  }

  /* ------------------------------------------------------------------ *
   * 4. Turn each price card into a real PayPal button.
   * ------------------------------------------------------------------ */
  function build() {
    if (!window.paypal || !window.paypal.Buttons) return;

    /* No agreement checkboxes on the page = no payments. Fail loudly and
       visibly rather than showing buttons that can never be unlocked. */
    if (!boxes().length) {
      console.error(
        "[SimplySteno PayPal] The three agreement checkboxes are missing from " +
        "this page, so no payment buttons will be shown. Restore the checkbox " +
        "HTML block above the price boxes and they will come back."
      );
      return;
    }

    var cards = document.querySelectorAll(".price_box");
    if (!cards.length) return;

    Array.prototype.forEach.call(cards, function (card, i) {
      var heading = card.querySelector(".elementor-image-box-title");
      if (!heading) return;

      var amount = priceFrom(heading.textContent);
      if (!amount) return;                       // not a payable card, skip
      var item = cleanName(heading.textContent);

      /* Kill the dead hermes link, keep the text. The old <a> pointed at an
         expired one-time checkout session — that is the bug we're fixing. */
      Array.prototype.forEach.call(
        card.querySelectorAll('a[href*="/webapps/hermes"]'),
        function (a) {
          var span = document.createElement("span");
          span.className = "ss-paypal-dead-link";
          span.innerHTML = a.innerHTML;
          a.parentNode.replaceChild(span, a);
        }
      );

      /* The old paypal.jpg image is now redundant — the real button
         carries PayPal's branding. Hide the empty figure it lived in. */
      var fig = card.querySelector(".elementor-image-box-img");
      if (fig && !fig.querySelector("a")) fig.style.display = "none";

      /* Mount point for the button, inside the card. */
      var mount = document.createElement("div");
      mount.className = "ss-paypal-button";
      mount.id = "ss-paypal-button-" + i;
      var content = card.querySelector(".elementor-image-box-content") || card;
      content.appendChild(mount);

      paypal.Buttons({
        style: { layout: "vertical", label: "buynow", shape: "rect", height: 40 },

        /* --- checkbox gate, PayPal's documented pattern ---------------- *
         * onInit disables the button until all three boxes are ticked;
         * onClick is the backstop if anything slips through.             */
        onInit: function (data, actions) {
          actions.disable();
          function sync() {
            if (allTicked()) { actions.enable(); hideWarning(); }
            else            { actions.disable(); }
          }
          boxes().forEach(function (c) { c.addEventListener("change", sync); });
          sync();
        },

        onClick: function (data, actions) {
          if (!allTicked()) {
            showWarning();
            return actions.reject();
          }
          return actions.resolve();
        },

        createOrder: function (data, actions) {
          return actions.order.create({
            purchase_units: [{
              description: item,
              amount: { value: amount, currency_code: CURRENCY }
            }],
            application_context: { shipping_preference: "NO_SHIPPING" }
          });
        },

        onApprove: function (data, actions) {
          return actions.order.capture().then(function (details) {
            var payer = (details.payer && details.payer.name &&
                         details.payer.name.given_name) || "";
            alert(
              "Thank you" + (payer ? ", " + payer : "") + "! Your payment for " +
              item + " went through.\n\nPayPal transaction ID: " + details.id +
              "\n\n" + THANK_YOU_MESSAGE
            );
          });
        },

        onError: function (err) {
          console.error("[SimplySteno PayPal]", err);
          alert(
            "Sorry — PayPal could not complete that payment.\n\n" +
            "Please try again, or email SpeedBuilders@hotmail.com and we'll " +
            "sort it out for you."
          );
        }
      }).render("#" + mount.id);
    });
  }

  /* ------------------------------------------------------------------ *
   * 5. Load the SDK, then build.
   * ------------------------------------------------------------------ */
  function loadSdk() {
    if (document.getElementById("ss-paypal-sdk")) return;
    var s = document.createElement("script");
    s.id = "ss-paypal-sdk";
    s.src = "https://www.paypal.com/sdk/js" +
            "?client-id=" + encodeURIComponent(PAYPAL_CLIENT_ID) +
            "&currency=" + encodeURIComponent(CURRENCY) +
            "&locale=" + encodeURIComponent(LOCALE) +
            "&intent=capture" +
            "&components=buttons" +
            "&enable-funding=paylater" +           // PayPal Credit
            "&disable-funding=" + encodeURIComponent(DISABLE_FUNDING);
    s.onload = build;
    s.onerror = function () {
      console.error("[SimplySteno PayPal] SDK failed to load.");
    };
    document.head.appendChild(s);
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", loadSdk);
  } else {
    loadSdk();
  }
})();
</script>
